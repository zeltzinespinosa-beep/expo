  const [tipo, setTipo] = useState<'general' | 'biologica' | 'quimica'>('general');

  const numeroParaTipo = () => {
    if (tipo === 'biologica') return NUMERO_BIOLOGICA;
    if (tipo === 'quimica') return NUMERO_QUIMICA;
    return NUMERO_GENERAL;
  };

  const llamar = async () => {
    const numero = numeroParaTipo();
    const url = `tel:${numero}`;

    try {
      const supported = await Linking.canOpenURL(url);
      if (!supported) {
        Alert.alert('Error', 'No se puede realizar la llamada desde este dispositivo.');
        return;
      }
      await Linking.openURL(url);
    } catch (err) {
      Alert.alert('Error', 'No se pudo iniciar la llamada.');

  };
